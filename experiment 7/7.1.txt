Implement a PL/SQL block with a cursor to fetch and display the Name and Salary of the
top 5 highest-paid employees from the Staff table. - 

DECLARE
    CURSOR emp_cursor IS
        SELECT Name, Salary
        FROM Staff
        ORDER BY Salary DESC;

    v_name   Staff.Name%TYPE;
    v_salary Staff.Salary%TYPE;
    v_count  NUMBER := 0;

BEGIN
    OPEN emp_cursor;

    LOOP
        FETCH emp_cursor INTO v_name, v_salary;
        EXIT WHEN emp_cursor%NOTFOUND OR v_count = 5;

        v_count := v_count + 1;

        DBMS_OUTPUT.PUT_LINE(
            'Name: ' || v_name || ' | Salary: ' || v_salary
        );
    END LOOP;

    CLOSE emp_cursor;
END;
